<script>
let currentRole = 'creator';
let creatorUsername = '';
 
// ─── SHARED STATE (persists across login switches in same session) ───
const state = {
  creators: {},       // { '@handle': { phase: 1, joinedAt: 'Just now' } }
  requests: [],       // [ { id, creator, context, phase, time, replied: bool } ]
  notifications: {},  // { '@handle': [ { id, title, script, time, read } ] }
};
 
// ─── ROLE SWITCH ───
function setRole(role) {
  currentRole = role;
  document.querySelectorAll('.role-tab').forEach((t,i) => {
    t.classList.toggle('active', (i===0&&role==='creator')||(i===1&&role==='admin'));
  });
  document.getElementById('creator-fields').style.display = role==='creator' ? 'block' : 'none';
  document.getElementById('admin-fields').style.display = role==='admin' ? 'block' : 'none';
}
 
// ─── LOGIN ───
function login() {
  if (currentRole === 'creator') {
    const name = document.getElementById('ig-input').value.trim();
    if (!name) { showToast('Enter your Instagram handle'); return; }
    creatorUsername = name.startsWith('@') ? name : '@' + name;
 
    // Register creator in state
    if (!state.creators[creatorUsername]) {
      state.creators[creatorUsername] = { phase: 1, joinedAt: 'Just now', hasRequest: false };
    }
 
    document.getElementById('creator-name-display').textContent = creatorUsername;
    const mobileDisplay = document.getElementById('creator-name-display-mobile');
    if (mobileDisplay) mobileDisplay.textContent = creatorUsername;
 
    // Reset request UI if they haven't submitted yet
    const hasRequest = state.requests.some(r => r.creator === creatorUsername);
    if (!hasRequest) {
      const inp = document.getElementById('script-request-input');
      const btn = document.querySelector('.btn-request');
      if (inp) { inp.disabled = false; inp.value = ''; }
      if (btn) { btn.textContent = 'Send Request →'; btn.style.opacity = '1'; btn.style.pointerEvents = 'auto'; }
    } else {
      const btn = document.querySelector('.btn-request');
      if (btn) { btn.textContent = 'Request sent ✓'; btn.style.opacity = '.5'; btn.style.pointerEvents = 'none'; }
    }
 
    switchScreen('screen-creator');
    refreshNotifBadge();
    renderNotifications();
 
  } else {
    const pw = document.getElementById('pw-input').value.trim();
    if (pw !== 'admin') { showToast('Incorrect password'); return; }
    switchScreen('screen-admin');
    renderAdminRequests();
    renderAdminCreators();
    refreshAdminBadge();
  }
}
 
// ─── LOGOUT ───
function logout() {
  document.getElementById('ig-input').value = '';
  document.getElementById('pw-input').value = '';
  // Reset creator nav to dashboard
  document.querySelectorAll('#screen-creator .page').forEach(p => p.classList.remove('active'));
  const dash = document.getElementById('page-dashboard');
  if (dash) dash.classList.add('active');
  document.querySelectorAll('#screen-creator .nav-item').forEach((n,i) => n.classList.toggle('active', i===0));
  switchScreen('screen-login');
  setRole('creator');
}
 
// ─── SCREEN SWITCH ───
function switchScreen(id) {
  document.querySelectorAll('.screen').forEach(s => {
    s.classList.remove('active'); s.style.display = 'none';
  });
  const target = document.getElementById(id);
  target.style.display = 'flex';
  setTimeout(() => target.classList.add('active'), 10);
}
 
// ─── CREATOR PAGE NAV ───
function showPage(name) {
  document.querySelectorAll('#screen-creator .page').forEach(p => p.classList.remove('active'));
  document.getElementById('page-' + name).classList.add('active');
  document.querySelectorAll('#screen-creator .nav-item').forEach(n => n.classList.remove('active'));
  event.currentTarget.classList.add('active');
  if (name === 'notifications') { renderNotifications(); markAllRead(); }
}
 
// ─── ADMIN PAGE NAV ───
function showAdminPage(name) {
  document.querySelectorAll('#screen-admin .page').forEach(p => p.classList.remove('active'));
  document.getElementById('page-' + name).classList.add('active');
  document.querySelectorAll('#screen-admin .nav-item').forEach(n => n.classList.remove('active'));
  event.currentTarget.classList.add('active');
  if (name === 'admin-requests') { renderAdminRequests(); refreshAdminBadge(); }
  if (name === 'admin-creators') renderAdminCreators();
}
 
// ─── CREATOR: SUBMIT SCRIPT REQUEST ───
function submitRequest() {
  const input = document.getElementById('script-request-input');
  const btn = document.querySelector('.btn-request');
  const context = input.value.trim();
  if (!context) { showToast('Add some context first'); return; }
 
  state.requests.push({
    id: Date.now(),
    creator: creatorUsername,
    context,
    phase: 'Phase 1 · Market Analysis',
    time: 'Just now',
    replied: false
  });
 
  if (state.creators[creatorUsername]) {
    state.creators[creatorUsername].hasRequest = true;
  }
 
  input.disabled = true;
  btn.textContent = 'Request sent ✓';
  btn.style.opacity = '.5';
  btn.style.pointerEvents = 'none';
 
  showToast('Request sent — check your notifications when the script is ready');
}
 
// ─── ADMIN: RENDER REQUESTS ───
function renderAdminRequests() {
  const list = document.getElementById('request-list');
  const empty = document.getElementById('requests-empty');
  const pending = state.requests.filter(r => !r.replied);
 
  if (pending.length === 0) {
    if (empty) empty.style.display = 'block';
    // Remove any rendered rows
    list.querySelectorAll('.request-row').forEach(r => r.remove());
    return;
  }
 
  if (empty) empty.style.display = 'none';
  list.querySelectorAll('.request-row').forEach(r => r.remove());
 
  pending.forEach(req => {
    const div = document.createElement('div');
    div.className = 'request-row';
    div.id = 'req-' + req.id;
    div.innerHTML = `
      <div class="request-meta">
        <div>
          <div class="request-from">${req.creator}</div>
          <div style="font-family:var(--font);font-size:10px;color:var(--gold);margin-top:2px">${req.phase}</div>
        </div>
        <div class="request-time">${req.time}</div>
      </div>
      <div class="request-msg">${req.context}</div>
      <textarea class="admin-reply-area" placeholder="Write your personalised script here..."></textarea>
      <button class="btn-send" onclick="sendScript(this, ${req.id}, '${req.creator}')">Send Script →</button>
    `;
    list.appendChild(div);
  });
}
 
// ─── ADMIN: RENDER CREATORS ───
function renderAdminCreators() {
  const list = document.getElementById('creator-list');
  const handles = Object.keys(state.creators);
 
  if (handles.length === 0) {
    list.innerHTML = `<div style="text-align:center;padding:48px 24px;background:var(--surface);border:1px solid var(--border);border-radius:6px;font-family:var(--serif);font-size:14px;color:var(--muted)">No creators yet. They'll appear here once they log in.</div>`;
    return;
  }
 
  list.innerHTML = handles.map(handle => {
    const c = state.creators[handle];
    const hasReq = state.requests.some(r => r.creator === handle && !r.replied);
    return `
      <div class="creator-row">
        <div>
          <div class="creator-name">${handle}</div>
          <div class="creator-meta">Phase ${c.phase} active${hasReq ? ' · Script request pending' : ''} · Joined ${c.joinedAt}</div>
        </div>
        <span class="phase-pill">Phase ${c.phase}</span>
        <span class="phase-status ${hasReq ? 'status-pending' : 'status-active'}">${hasReq ? 'Request pending' : 'Active'}</span>
        <button class="btn-unlock" onclick="unlockPhase(this,'${handle}')">Unlock Phase ${c.phase + 1}</button>
      </div>
    `;
  }).join('');
}
 
// ─── ADMIN: SEND SCRIPT → CREATOR INBOX ───
function sendScript(btn, reqId, creator) {
  const row = document.getElementById('req-' + reqId);
  const reply = row.querySelector('.admin-reply-area').value.trim();
  if (!reply) { showToast('Write a script first'); return; }
 
  // Mark request as replied
  const req = state.requests.find(r => r.id === reqId);
  if (req) req.replied = true;
 
  // Push to creator notification store
  if (!state.notifications[creator]) state.notifications[creator] = [];
  state.notifications[creator].unshift({
    id: Date.now(),
    title: 'Your Personalised Script is Ready',
    phase: req ? req.phase : 'Phase 1',
    script: reply,
    time: 'Just now',
    read: false
  });
 
  row.style.opacity = '.4';
  row.style.pointerEvents = 'none';
  btn.textContent = 'Sent ✓';
 
  refreshAdminBadge();
  showToast('Script sent to ' + creator);
}

function mobileShowPage(name, el) {

  if (name === 'admin-requests') { renderAdminRequests(); refreshAdminBadge(); }
  if (name === 'admin-creators') renderAdminCreators();
}
 
// ─── BADGE ───
function refreshAdminBadge() {
  const pending = state.requests.filter(r => !r.replied).length;
  const badge = document.getElementById('request-badge');
  const mobileBadge = document.getElementById('mobile-request-badge');
  if (pending > 0) {
    badge.textContent = pending; badge.style.display = 'inline-block';
    if (mobileBadge) { mobileBadge.textContent = pending; mobileBadge.style.display = 'block'; }
  } else {
    badge.style.display = 'none';
    if (mobileBadge) mobileBadge.style.display = 'none';
  }
}
 
// ─── CREATOR: RENDER NOTIFICATIONS ───
function renderNotifications() {
  const list = document.getElementById('notif-list');
  const empty = document.getElementById('notif-empty');
  const notifs = state.notifications[creatorUsername] || [];
 
  if (notifs.length === 0) {
    empty.style.display = 'block';
    list.innerHTML = '';
    return;
  }
 
  empty.style.display = 'none';
  list.innerHTML = notifs.map(n => `
    <div class="notif-card ${n.read ? '' : 'unread'}" id="notif-${n.id}">
      <div class="notif-card-header">
        <div class="notif-card-header-left">
          <div class="notif-title">
            ${!n.read ? '<span class="unread-dot"></span>' : ''}
            ${n.title}
          </div>
          <div class="notif-meta">${n.phase} &nbsp;·&nbsp; ${n.time}</div>
        </div>
      </div>
      <div class="notif-card-body">${n.script}</div>
      <div class="notif-card-footer">
        <button class="btn-copy-script" onclick="copyScript(this, ${n.id})">Copy Script</button>
        ${!n.read ? `<button class="btn-mark-read" onclick="markOneRead(${n.id})">Mark as read</button>` : '<span style="font-family:var(--font);font-size:11px;color:var(--muted);padding:9px 4px">✓ Read</span>'}
      </div>
    </div>
  `).join('');
}
 
function copyScript(btn, id) {
  const card = document.getElementById('notif-' + id);
  const text = card.querySelector('.notif-card-body').textContent.trim();
  navigator.clipboard.writeText(text);
  btn.textContent = 'Copied ✓';
  showToast('Script copied to clipboard');
  setTimeout(() => btn.textContent = 'Copy Script', 2000);
}
 
function markOneRead(id) {
  const notifs = state.notifications[creatorUsername] || [];
  const n = notifs.find(x => x.id === id);
  if (n) n.read = true;
  renderNotifications();
  refreshNotifBadge();
}
 
function markAllRead() {
  const notifs = state.notifications[creatorUsername] || [];
  notifs.forEach(n => n.read = true);
  refreshNotifBadge();
}
 
// ─── CREATOR BADGE ───
function refreshNotifBadge() {
  const notifs = state.notifications[creatorUsername] || [];
  const unread = notifs.filter(n => !n.read).length;
  const badge = document.getElementById('notif-badge');
  const mobileBadge = document.getElementById('mobile-notif-badge');
  if (unread > 0) {
    badge.textContent = unread; badge.style.display = 'inline-block';
    if (mobileBadge) { mobileBadge.textContent = unread; mobileBadge.style.display = 'block'; }
  } else {
    badge.style.display = 'none';
    if (mobileBadge) mobileBadge.style.display = 'none';
  }
}

}
 
// ─── CTA COPY ───
function copyCTA(btn, variant) {
  const texts = {
    A: "I'm finalising a [SYSTEM/PROGRAM] for [AUDIENCE] right now. Everyone who fills out the form in my bio gets first access + a free guide.",
    B: "Building something for [AUDIENCE] in the next few weeks. Form in my bio if you want early access.",
    C: "Closing the early-access list soon. Bio link if you want a spot."
  };
  navigator.clipboard.writeText(texts[variant]);
  btn.textContent = 'Copied ✓';
  setTimeout(() => btn.textContent = 'Copy text', 2000);
}
 
// ─── UNLOCK PHASE ───
function unlockPhase(btn, creator) {
  if (state.creators[creator]) {
    state.creators[creator].phase = Math.min(4, state.creators[creator].phase + 1);
  }
  btn.textContent = 'Unlocked ✓';
  btn.style.opacity = '.5';
  btn.style.pointerEvents = 'none';
  showToast('Phase unlocked for ' + creator);
  renderAdminCreators();
}
 
// ─── TOAST ───
function showToast(msg) {
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.classList.add('show');
  setTimeout(() => t.classList.remove('show'), 3200);
}
